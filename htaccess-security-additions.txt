# --- CRITICAL: Disable error display in production ---
php_value display_errors 0
php_value log_errors 1
php_value error_log /home/u143213086/logs/php_errors.log

# --- Block direct access to middleware files ---
<FilesMatch "^(auth|csrf|rate-limiter)\.php$">
    <If "%{REQUEST_URI} =~ m#/middleware/#">
        Require all denied
    </If>
</FilesMatch>

# --- Block access to sensitive files ---
<FilesMatch "\.(sql|log|env|bak|config)$">
    Require all denied
</FilesMatch>

# --- Block access to .git directory ---
RedirectMatch 404 /\.git

# --- Security headers (supplement Cloudflare) ---
<IfModule mod_headers.c>
    # Prevent MIME-type sniffing
    Header always set X-Content-Type-Options "nosniff"
    # Prevent clickjacking
    Header always set X-Frame-Options "SAMEORIGIN"
    # XSS protection (legacy browsers)
    Header always set X-XSS-Protection "1; mode=block"
    # Referrer policy
    Header always set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>
